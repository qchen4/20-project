     22  `timescale 1ns/1ns
     23  
     24  `include "sim_define.v"
     25  
     26  module tb;
     27  
     28    logic         rstn, clk   ;
     29    logic         m_hreadyout ;
     30    logic         m_hresp     ;
     31  
     32    logic         s_hresp     ;
     33    logic [31:0]  dut_mon_sig ;   //singal inside DUT to be monitor
     34  
     35    always #(10/2.0) begin
     36      clk = ~clk;
     37    end
     38  
     39    // System Clock and Reset
     40    initial begin
     41      rstn = 0; clk = 0;
     42  
     43      //--- first reset
     44      repeat(3) @(posedge clk);
     45      #1; rstn = 0;
     46      repeat(3) @(posedge clk);
     47      #1; rstn = 1;
     48      
     49      /*
     50      repeat(40) @(posedge clk);
     51     
     52      //--- second reset
     53      #1; rstn = 0;
     54      repeat(3) @(posedge clk);
     55      #1; rstn = 1;
     56      */
     57    end
     58  
     59      //interface instance
     60      yuu_ahb_master_interface  m_ahb(.dut_mon(dut_mon_sig));
     61      yuu_ahb_slave_interface   s_ahb();
     62  
     63      //DUT internal signal monitor infterface
     64      i_cache_monitor_interface dut_mon(.clk(clk), .rstn(rstn));
     65  
     66      //m-ahb/s-ahb interface signal connection
     67      assign  m_ahb.hclk      = clk;
     68      assign  m_ahb.hreset_n  = rstn;
     69      assign  m_ahb.hexokay   = 1'b0; 
     70      assign  #1 m_ahb.hready_i= m_hreadyout;
     71      assign  #1 m_ahb.hresp  = {1'b0, m_hresp};
     72  
     73      assign  s_ahb.hclk      = clk;
     74      assign  s_ahb.hreset_n  = rstn;
     75      assign  #1 s_ahb.hready_i = s_ahb.hready_o;
     76      assign  #1 s_hresp      = s_ahb.hresp[0];
     77      assign  s_ahb.hmaster   = 'd0;
     78      assign  s_ahb.hprot_emt = 'd0;
     79      assign  s_ahb.hnonsec   = 1'b0;
     80      assign  s_ahb.excl_write_enable = 1'b0;
     81  
     82      //DUT monitor signals
     83      wire    cache_init  ;
     84      wire    m_wrap_dphase;
     85      wire    cache_line_wr;
     86  
     87      assign  cache_init      = tb.u_icache_ahb.u_icache_tag.cache_init;
     88      assign  m_wrap_dphase   = tb.u_icache_ahb.m_wrap_dphase;
     89      assign  cache_line_wr   = tb.u_icache_ahb.u_icache_dmem.cache_line_wr;
     90  
     91      assign  dut_mon_sig = {28'h0,
     92                             1'h0, cache_line_wr, m_wrap_dphase, cache_init};
     93      
     94      assign  dut_mon.hit_chk_point= tb.u_icache_ahb.u_icache_tag.hit_chk_point;
     95      assign  dut_mon.tag_info_used= tb.u_icache_ahb.u_icache_tag.tag_info_used;
     96      assign  dut_mon.chk_haddr_d  = tb.u_icache_ahb.u_icache_tag.chk_haddr_d;
     97      assign  dut_mon.w0_hit       = tb.u_icache_ahb.u_icache_tag.w0_hit;
     98      assign  dut_mon.w1_hit       = tb.u_icache_ahb.u_icache_tag.w1_hit;
     99      assign  dut_mon.w2_hit       = tb.u_icache_ahb.u_icache_tag.w2_hit;
     100      assign  dut_mon.w3_hit       = tb.u_icache_ahb.u_icache_tag.w3_hit;
     101      assign  dut_mon.fill_hit     = tb.u_icache_ahb.u_icache_tag.fill_hit;
     102      assign  dut_mon.cache_miss   = tb.u_icache_ahb.u_icache_tag.cache_miss;
     103      
     104  
     105      //DUT instance
     106      icache_ahb #(.index_bits(`INDEX_BITS) ) u_icache_ahb(  //4KB i-cache
     107  	//--- upstream ahb-lite slave interface
     108  	.s_hsel			(1'b1           ),
     109  	.s_haddr		(m_ahb.haddr	),
     110  	.s_hburst		(m_ahb.hburst	),
     111  	.s_htrans		(m_ahb.htrans	),
     112  	.s_hsize		(m_ahb.hsize	),
     113  	.s_hprot		(m_ahb.hprot	),
     114  	.s_hwrite		(m_ahb.hwrite	),
     115  	.s_hwdata		(m_ahb.hwdata	),
     116      .s_hmastlock    (m_ahb.hmastlock),
     117  	.s_hready		(m_ahb.hready_i ),
     118  	.s_hreadyout	(m_hreadyout    ),
     119  	.s_hrdata		(m_ahb.hrdata	),
     120  	.s_hresp		(m_hresp	    ),
     121  
     122      //--- downstream ahb-master interface
     123  	.m_hsel			(s_ahb.hsel     ),
     124  	.m_haddr		(s_ahb.haddr	),
     125  	.m_hburst		(s_ahb.hburst	),
     126  	.m_htrans		(s_ahb.htrans	),
     127  	.m_hsize		(s_ahb.hsize	),
     128  	.m_hprot		(s_ahb.hprot	),
     129  	.m_hwrite		(s_ahb.hwrite	),
     130  	.m_hwdata		(s_ahb.hwdata	),
     131      .m_hmastlock    (s_ahb.hmastlock),
     132  	.m_hready		(s_ahb.hready_i ),
     133  	.m_hrdata		(s_ahb.hrdata   ),
     134  	.m_hresp		(s_hresp        ),
     135  
     136      .clk            (clk            ),
     137      .rstn           (rstn           )  
     138      );
     139  
     140      //test program
     141      test u_test(.m_ahb(m_ahb), .s_ahb(s_ahb), .dut_mon(dut_mon));
     142  
     143      m_ahb_cyc_coverage m_cyc_coverage(.ahb_io(m_ahb));
     144  
     145  endmodule : tb
