     4  class i_cache_coverage extends uvm_component;
     5  
     6      `uvm_component_utils(i_cache_coverage)    
     7      uvm_analysis_imp #(i_cache_coverage_item, i_cache_coverage) scb2cov;
     8  
     9  	//Cover the current item and cache status
     10  	covergroup cover_item with function sample(i_cache_coverage_item  item);
     11  		option.per_instance = 1;
     12  
     13          //---1: m-ahb trans cover
     14  		ahb_rw : coverpoint item.ahb_item.direction {
     15  			type_option.comment = "m-ahb read/write access.";
     16  		}
     17          
     18  		ahb_rw_trans : coverpoint item.ahb_item.direction {
     19  			type_option.comment = "m-ahb read/write access change.";
     20              bins ahb_rw_trans[] = (READ, WRITE => READ, WRITE);
     21  		}
     22  
     23          //R/W addr toggle
     24          //                   name                data                       condition
     25          `cover_val_32_bits(raddr_toggle, item.ahb_item.start_address, iff(item.ahb_item.direction == READ))
     26          `cover_val_32_bits(waddr_toggle, item.ahb_item.start_address, iff(item.ahb_item.direction == WRITE))
     27  
     28          ahb_addr_max_min : coverpoint item.ahb_item.start_address {
     29  			type_option.comment = "m-ahb max/min haddr.";
     30              bins min_haddr = {32'h0};
     31              bins max_haddr = {32'hffff_ffff};
     32  		}
     33  
     34          ahb_prot0 : coverpoint item.ahb_item.prot0 {
     35              type_option.comment = "m-ahb instruction or data trans.";
     36          }
     37  
     38          ahb_prot3 : coverpoint item.ahb_item.prot3 {
     39              type_option.comment = "m-ahb non-cacheable or cacheable trans.";
     40          }
     41  
     42          ahb_hsize : coverpoint item.ahb_item.size {
     43              type_option.comment = "m-ahb hsize.";
     44              bins hsize[] = {SIZE8, SIZE16, SIZE32};
     45          }
     46  
     47          ahb_hburst: coverpoint item.ahb_item.burst {
     48              type_option.comment = "m-ahb hburst.";
     49              bins single = {SINGLE};
     50          }
     51  
     52          ahb_addr_offset : coverpoint item.ahb_item.address[0][1:0] {
     53              type_option.comment = "m-ahb address byte offset.";
     54              bins haddr_offset[] = {0, 1, 2, 3};
     55          }
     56  
     57          ahb_trans_cache : cross cache_init, ahb_prot0, ahb_rw, ahb_prot3, ahb_hsize {
     58              ignore_bins no_write_on_inst = binsof(ahb_prot0) intersect {OPCODE_FETCH} && binsof(ahb_rw) intersect {WRITE};
     59          }
     60  
     61          ahb_trans_all : cross cache_init, ahb_rw, ahb_hsize, ahb_addr_offset {
     62              ignore_bins hsize_32b_ignore = binsof(ahb_hsize) intersect {SIZE32} && binsof(ahb_addr_offset) intersect {1,2,3};
     63              ignore_bins hsize_16b_ignore = binsof(ahb_hsize) intersect {SIZE16} && binsof(ahb_addr_offset) intersect {1,3};
     64          }
     65  
     66          //R/W data_toggle
     67          //                   name                data                       condition
     68          `cover_val_32_bits(wdata_toggle, item.ahb_item.data[0], iff(item.ahb_item.direction == WRITE))
     69          `cover_val_32_bits(rdata_toggle, item.ahb_item.data[0], iff(item.ahb_item.direction == READ))
     70  
     71          ahb_hprot : coverpoint {item.ahb_item.prot3, item.ahb_item.prot2, item.ahb_item.prot1, item.ahb_item.prot0} {
     72              type_option.comment = "m-ahb hprot.";
     73          }
     74  
     75          ahb_hresp : coverpoint item.ahb_item.response[0] {
     76              type_option.comment = "m-ahb hresp.";
     77          }
     78  
     79          ahb_hresp_rw: cross cache_init, ahb_rw, ahb_hresp;
     80  
     81          ahb_hmastlock : coverpoint item.ahb_item.lock {
     82              type_option.comment = "m-ahb hmastlock.";
     83          }
     84  
     85          ahb_hmastlock_rw: cross cache_init, ahb_rw, ahb_hmastlock;
     86  
     87          idle_between_burst: coverpoint item.ahb_item.idle_delay {
     88              bins idle_cycles[] = {[0:7]};
     89          }
     90  
     91          //---2: cache status check
     92          cache_init : coverpoint item.cache_init {
     93              type_option.comment = "Cache initial or working status.";
     94              bins cache_init = {1};
     95              bins cache_work = {0};
     96          }
     97  
     98          cache_miss : coverpoint item.miss iff((item.cache_init == 0) && (item.i_trans) && (item.ahb_item.prot3)){
     99              type_option.comment = "cache miss.";
     100          }
     101          
     102          cache_tag_hit : coverpoint item.tag_hit iff((item.cache_init == 0) && (item.i_trans) && (item.ahb_item.prot3)){
     103              type_option.comment = "cache tag_hit[3:0] of all ways.";
     104              bins tag_hit[] = {4'b0000, 4'b0001, 4'b0010, 4'b0100, 4'b1000}; //one-hot encoding
     105          }
     106  
     107          cache_tag_hit_flag : coverpoint item.tag_hit_flag iff((item.cache_init == 0) && (item.i_trans) && (item.ahb_item.prot3)){
     108              type_option.comment = "cache tag_hit_flag.";
     109          }
     110  
     111          cache_multi_hit : coverpoint item.tag_hit_flag iff((item.cache_init == 0) && (item.i_trans) && (item.ahb_item.prot3)){
     112              type_option.comment = "cache tag hit multiple times.";
     113              bins hit2 = (1 => 1);
     114              bins hit3 = (1 => 1[*2]);
     115              bins hit4 = (1 => 1[*3]);
     116              bins hit5 = (1 => 1[*4]);
     117          }
     118  
     119          cache_hit_fill_line : coverpoint item.fill_hit iff((item.cache_init == 0) && (item.i_trans) && (item.ahb_item.prot3)){
     120              type_option.comment = "cache hit filling line.";
     121          }
     122  
     123          cache_miss_then_tag_hit : coverpoint {item.tag_hit_flag, item.miss, item.m_wrap_dphase, item.cache_line_wr} iff((item.cache_init == 0) && (item.i_trans) && (item.ahb_item.prot3)){
     124              type_option.comment = "cache miss followed by tag hit.";
     125              bins miss_then_tag_hit = (4'b0100 => 4'b1010);
     126          }
     127  
     128          cache_miss_then_tag_hit_multi : coverpoint {item.tag_hit_flag, item.miss, item.m_wrap_dphase, item.cache_line_wr} iff((item.cache_init == 0) && (item.i_trans) && (item.ahb_item.prot3)){
     129              type_option.comment = "cache miss followed by tag hit multiple times.";
     130              bins miss_then_tag_hit_2 = (4'b0100 => 4'b1010[*2]);
     131              bins miss_then_tag_hit_3 = (4'b0100 => 4'b1010[*3]);
     132          }
     133  
     134          cache_trans_when_filling : coverpoint {item.ahb_item.prot0, item.ahb_item.prot3, item.ahb_item.direction} iff((item.cache_init == 0) && (item.m_wrap_dphase) && (!item.cache_line_wr)){
     135              type_option.comment = "m-ahb trans when reading missing cache line.";
     136              bins data_read = {3'b110, 3'b100};
     137              bins data_write= {3'b111, 3'b101};
     138              bins cacheable_i_read = {3'b010};
     139              bins noncacheable_i_read = {3'b000};
     140          }
     141  
     142          cache_trans_when_write_dmem : coverpoint {item.ahb_item.prot0, item.ahb_item.prot3, item.ahb_item.direction} iff((item.cache_init == 0) && (item.m_wrap_dphase) && (item.cache_line_wr)){
     143              type_option.comment = "m-ahb trans when write missing cache line to data mem.";
     144              //bins data_read = {3'b110, 3'b100};
     145              //bins data_write= {3'b111, 3'b101};
     146              bins cacheable_i_read = {3'b010};
     147              //bins noncacheable_i_read = {3'b000};
     148          }
     149  
     150          cache_miss_then_miss : coverpoint {item.miss, item.m_wrap_dphase, item.cache_line_wr} iff((item.cache_init == 0) && (item.i_trans) && (item.ahb_item.prot3)){
     151              type_option.comment = "cache miss followed by cache miss.";
     152              bins miss_then_miss = (3'b100 => 3'b110);
     153          }
     154  
     155          cache_hit_fill_index : coverpoint (item.hit_fill_index) iff(item.tag_hit != 0){
     156              type_option.comment = "cache tag hit, hit index == index of current filling line.";
     157          }
     158  
     159          cache_index : coverpoint item.a_index iff((item.cache_init == 0) && (item.i_trans) && (item.ahb_item.prot3)){
     160              type_option.comment = "cache index.";
     161              bins cache_index[] = {[0:((1<<`INDEX_BITS)-1)]};
     162          }
     163  
     164          cache_hit_miss_all_index : cross cache_tag_hit, cache_index;
     165      endgroup
     166

     179          super.connect_phase(phase);
     180      endfunction : connect_phase

     202  function void i_cache_coverage::icache_status_print(i_cache_coverage_item item);
     203      $display("At time %0t, print icache_status values:", $time);
     204      $display("cahce_init: %0h", item.cache_init);
     205      $display("m_wrap_dphase: %0h", item.m_wrap_dphase);
     206      $display("cache_line_wr: %0h", item.cache_line_wr);
     207      $display("i_trans: %0h", item.i_trans);
     208      $display("tag_hit: %0h", item.tag_hit);
     209      $display("tag_hit_flag: %0h", item.tag_hit_flag);
     210      $display("miss: %0h", item.miss);
     211      $display("replace_way: %0h", item.replace_way);
     212      $display("a_index: %0h", item.a_index);
     213      $display("hit_fill_index: %0h", item.hit_fill_index);
     214  endfunction
     215  
     216  function void i_cache_coverage::icache_item_print(yuu_ahb_item item);
     217      $display("At time %0t, print ahb trans item values:", $time);
     218      $display("addr : %0h", item.address[0]);
     219      $display("direction : %0h", item.direction);
     220      $display("hsize : %0h", item.size);
     221      $display("prot0: %0h", item.prot0);
     222      $display("prot3: %0h", item.prot3);
     223  endfunction
