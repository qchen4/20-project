     5  `ifndef GUARD_YUU_AHB_MASTER_INTERFACE_SVI
     6  `define GUARD_YUU_AHB_MASTER_INTERFACE_SVI
     7  
     8  // Interface: yuu_ahb_master_interface
     9  // AHB master bus interface 
     10  interface yuu_ahb_master_interface(input logic [31:0] dut_mon);
     11    logic                               hclk;
     12    logic                               hreset_n;
     13    logic [`YUU_AHB_MAX_ADDR_WIDTH-1:0] haddr;
     14    logic [1:0]                         htrans;
     15    logic [2:0]                         hburst;
     16    logic                               hwrite;
     17    logic [2:0]                         hsize;
     18    logic [`YUU_AHB_MAX_DATA_WIDTH-1:0] hwdata;
     19    logic [`YUU_AHB_MAX_DATA_WIDTH-1:0] hrdata;
     20    logic                               hready_i;
     21    logic [1:0]                         hresp;
     22  
     23    logic [3:0]                         hprot;
     24    logic [6:3]                         hprot_emt;
     25    logic [3:0]                         hmaster;
     26    logic                               hmastlock;
     27    logic                               hnonsec;
     28    logic                               hexcl;
     29    logic                               hexokay;
     30    
     31    logic [31:0]                        upper_byte_lane;
     32    logic [31:0]                        lower_byte_lane;
     33  
     34    //--- signals used to help monitor and driver
     35    logic                               data_phase;   //data phase of nonseq and seq trans beat
     36    logic [31:0]                        busy_cnt;     //busy cycle before a burst beat
     37    logic [31:0]                        idle_cnt;     //idle cycle before first beat of a burst
     38  
     39  
     40    clocking cb @(posedge hclk);
     41    endclocking
     42  
     43    task wait_cycle();
     44      @(posedge hclk);
     45    endtask
     46  
     47  
     48  
     49    clocking drv_cb @(posedge hclk);
     50      default input #`YUU_AHB_MASTER_SETUP_TIME output #`YUU_AHB_MASTER_HOLD_TIME;
     51  
     52      output  haddr;
     53      output  htrans;
     54      output  hburst;
     55      output  hwrite;
     56      output  hsize;
     57      output  hwdata;
     58      output  hprot;
     59      output  hprot_emt;
     60      output  hmaster;
     61      output  hmastlock;
     62      output  hnonsec;
     63      output  hexcl;
     64  
     65      input   hrdata;
     66      input   hresp;
     67      input   hready_i;
     68      input   hexokay;
     69  
     70      output  upper_byte_lane;
     71      output  lower_byte_lane;
     72    endclocking
     73  
     74    clocking mon_cb @(posedge hclk);
     75      default input #`YUU_AHB_MASTER_SETUP_TIME output #`YUU_AHB_MASTER_HOLD_TIME;
     76  
     77      input   haddr;
     78      input   htrans;
     79      input   hburst;
     80      input   hwrite;
     81      input   hsize;
     82      input   hwdata;
     83      input   hprot;
     84      input   hprot_emt;
     85      input   hmaster;
     86      input   hmastlock;
     87      input   hnonsec;
     88      input   hexcl;
     89      input   hexokay;
     90  
     91      input   hrdata;
     92      input   hresp;
     93      input   hready_i;
     94  
     95      input   upper_byte_lane;
     96      input   lower_byte_lane;
     97  
     98      input   dut_mon;
     99      
     100      //--- signals used to help monitor and driver
     101      input   data_phase;
     102      input   idle_cnt;
     103      input   busy_cnt;
     104  
     105    endclocking
     106  
     107  
     108    modport drv_mp (
     109      /*
     110      output  haddr,
     111      output  htrans,
     112      output  hburst,
     113      output  hwrite,
     114      output  hsize,
     115      output  hwdata,
     116      output  hprot,
     117      output  hmaster,
     118      output  hmastlock,
     119      output  hnonsec,
     120      output  hexcl,
     121  
     122      input   hrdata,
     123      input   hresp,
     124      input   hready_i,
     125      input   hexokay,
     126      */
     127      input   hreset_n,
     128      clocking drv_cb
     129    );
     130  
     131    modport mon_mp (
     132      /*
     133      input   haddr,
     134      input   htrans,
     135      input   hburst,
     136      input   hwrite,
     137      input   hsize,
     138      input   hwdata,
     139      input   hprot,
     140      input   hmaster,
     141      input   hmastlock,
     142      input   hnonsec,
     143      input   hexcl,
     144  
     145      input   hrdata,
     146      input   hresp,
     147      input   hready_i,
     148      input   hexokay,
     149      */
     150      input   hreset_n,
     151      clocking mon_cb
     152    );
     153  
     154    // Analyze signal used to help monitor and driver
     155    `define AHBM_IF_TRANS_IDLE    0
     156    `define AHBM_IF_TRANS_BUSY    1
     157    `define AHBM_IF_TRANS_NONSEQ  2
     158    `define AHBM_IF_TRANS_SEQ     3
     159  
     160  
     161    always @(posedge hclk or negedge hreset_n)
     162      if(!hreset_n)
     163          data_phase  <= 1'b0;
     164      else if(((htrans == `AHBM_IF_TRANS_NONSEQ) || (htrans == `AHBM_IF_TRANS_SEQ)) && hready_i)
     165          data_phase  <= 1'b1;
     166      else if(hready_i)
     167          data_phase  <= 1'b0;
     168  
     169    always @(posedge hclk or negedge hreset_n)
     170      if(!hreset_n)
     171          idle_cnt    <= 'd0;
     172      else if(htrans == `AHBM_IF_TRANS_IDLE)
     173          idle_cnt    <= idle_cnt + 'd1;
     174      else if(hready_i)
     175          idle_cnt    <= 'd0;
     176  
     177    always @(posedge hclk or negedge hreset_n)
     178      if(!hreset_n)
     179          busy_cnt    <= 'd0;
     180      else if(htrans == `AHBM_IF_TRANS_BUSY)
     181          busy_cnt    <= busy_cnt + 'd1;
     182      //if idle trans after busy trans, will keep the value of busy_cnt
     183      else if(hready_i && ((htrans == `AHBM_IF_TRANS_NONSEQ) || (htrans == `AHBM_IF_TRANS_SEQ)))
     184          busy_cnt    <= 'd0;
     185  
     186  endinterface
     187  
     188  typedef virtual yuu_ahb_master_interface        yuv_ahb_mas_vif;
     189  
     190  `endif
